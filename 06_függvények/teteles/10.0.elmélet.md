            legnagyobb = szam
    return legnagyobb

    legidosebb_nev = ""

        if aktiv == True and kor > max_kor:
            max_kor = kor
            legidosebb_nev = nev
    return legidosebb_nev, max_kor

# Alap példa
def novekvo_rendezes(szamok):
    rendezett = list(szamok)
    for i in range(len(rendezett)):
        for j in range(i + 1, len(rendezett)):
            if rendezett[i] > rendezett[j]:
                temp = rendezett[i]
                rendezett[i] = rendezett[j]
                rendezett[j] = temp
    return rendezett

lista = [42, 12, 88, 3, 25]
sorbarendezett_szamok = novekvo_rendezes(lista)
print(f"Növekvő sorrend: {sorbarendezett_szamok}")


# Több feltételes példa
def versenyzok_rendezese(versenyzok):
    rendezett = list(versenyzok)
    for i in range(len(rendezett)):
        for j in range(i + 1, len(rendezett)):
            pont_i, kor_i = rendezett[i][1], rendezett[i][2]
            pont_j, kor_j = rendezett[j][1], rendezett[j][2]
            
            if pont_j > pont_i or (pont_i == pont_j and kor_j < kor_i):
                temp = rendezett[i]
                rendezett[i] = rendezett[j]
                rendezett[j] = temp
    return rendezett

adatok = [
    ["Péter", 85, 20],
    ["Anna", 95, 22],
    ["Gábor", 85, 18],
    ["Éva", 95, 19]
]

vegeredmeny = versenyzok_rendezese(adatok)

for v in vegeredmeny:
    print(f"{v[0]} - Pont: {v[1]}, Kor: {v[2]}")
